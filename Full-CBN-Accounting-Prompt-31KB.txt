<?php
namespace App\Services;

use App\Models\JournalEntry;
use App\Models\LedgerAccount;
use App\Models\PeriodClose;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

class LedgerService
{
  public static function post(
    string $debitCode,
    string $creditCode,
    float $amount,
    Model $reference,
    string $narration,
    ?\App\Models\User $postedBy = null,
    ?string $txnDate = null
  ): JournalEntry {
    if ($amount <= 0) throw new \Exception("Amount must be >0");
    if ($debitCode === $creditCode) throw new \Exception("Debit and Credit cannot be same");
    
    $txnDate = $txnDate ?? now()->toDateString();
    $period = substr($txnDate, 0, 7); // 2026-07
    
    if (PeriodClose::where('period', $period)->where('is_closed', true)->exists()) {
      throw new \Exception("Period $period is closed. Cannot post.");
    }

    return DB::transaction(function () use ($debitCode, $creditCode, $amount, $reference, $narration, $postedBy, $txnDate, $period) {
      $debitAccount = LedgerAccount::where('code', $debitCode)->where('is_active', true)->lockForUpdate()->firstOrFail();
      $creditAccount = LedgerAccount::where('code', $creditCode)->where('is_active', true)->lockForUpdate()->firstOrFail();

      // Prevent manual entry to control accounts unless from control module
      // Simplified check: if control account and reference not from module, block
      // Add your module check logic here

      $lastEntry = JournalEntry::orderBy('id', 'desc')->lockForUpdate()->first();
      $prevHash = $lastEntry?->hash ?? 'GENESIS';
      $uuid = (string) Str::uuid();
      $entryNumber = self::generateEntryNumber();
      $hash = hash('sha256', $uuid.$debitCode.$creditCode.$amount.$prevHash.now()->toISOString());

      $entry = JournalEntry::create([
        'uuid' => $uuid,
        'entry_number' => $entryNumber,
        'txn_date' => $txnDate,
        'debit_account_id' => $debitAccount->id,
        'credit_account_id' => $creditAccount->id,
        'amount' => $amount,
        'reference_type' => get_class($reference),
        'reference_id' => $reference->id,
        'reference_number' => $reference->reference ?? $reference->loan_number ?? $reference->id,
        'narration' => $narration,
        'posted_by' => $postedBy?->id ?? auth()->id(),
        'period' => $period,
        'prev_hash' => $prevHash,
        'hash' => $hash,
        'ip_address' => request()->ip(),
        'user_agent' => request()->userAgent(),
      ]);

      return $entry;
    });
  }

  public static function postSplit(array $entries, Model $reference, string $narration): array
  {
    // For loan repayment split: [['debit'=>'1002','credit'=>'1201','amount'=>15000], ['debit'=>'1002','credit'=>'4001','amount'=>3500]]
    // All in one transaction, same reference
    return DB::transaction(function () use ($entries, $reference, $narration) {
      $results = [];
      foreach ($entries as $e) {
        $results[] = self::post($e['debit'], $e['credit'], $e['amount'], $reference, $narration." - {$e['credit']}");
      }
      return $results;
    });
  }

  private static function generateEntryNumber(): string
  {
    $year = date('Y');
    $count = JournalEntry::whereYear('txn_date', $year)->count() + 1;
    return 'JE/'.$year.'/'.str_pad($count, 6, '0', STR_PAD_LEFT);
  }

  public static function reverse(JournalEntry $entry, string $reason, $user): JournalEntry
  {
    return DB::transaction(function () use ($entry, $reason, $user) {
      if ($entry->is_reversed) throw new \Exception("Already reversed");
      
      // Create reversal entry swapping debit/credit
      $reversal = self::post(
        debitCode: $entry->creditAccount->code,
        creditCode: $entry->debitAccount->code,
        amount: $entry->amount,
        reference: $entry->reference,
        narration: "REVERSAL of {$entry->entry_number}: $reason",
        postedBy: $user,
        txnDate: now()->toDateString()
      );

      $entry->update(['is_reversed' => true, 'reversed_by_id' => $reversal->id, 'reversal_reason' => $reason]);
      
      return $reversal;
    });
  }
}

<?php
namespace App\Services;
use App\Models\Loan;
use App\Models\LoanRepaymentSchedule;
use Carbon\Carbon;

class ProvisioningService
{
  public static function calculate(): array
  {
    $loans = Loan::whereIn('status', ['repaying','arrears','defaulted'])->with('schedules')->get();
    $totalProvision = 0;
    $results = [];

    foreach ($loans as $loan) {
      $overdueSchedules = $loan->schedules->where('status','overdue');
      $daysPastDue = 0;
      if ($overdueSchedules->isNotEmpty()) {
        $oldestDue = $overdueSchedules->min('due_date');
        $daysPastDue = Carbon::parse($oldestDue)->diffInDays(now());
      } elseif ($loan->status === 'repaying') {
        $daysPastDue = 0;
      }

      $rate = match(true) {
        $daysPastDue <= 30 => 0.01, // Performing - 1%
        $daysPastDue <= 60 => 0.25, // Pass & Watch - 25%
        $daysPastDue <= 90 => 0.50, // Substandard - 50%
        $daysPastDue <= 180 => 0.75, // Doubtful - 75%
        default => 1.00, // Lost - 100%
      };

      $outstanding = $loan->principal - $loan->repayments()->sum('principal_portion');
      $provision = $outstanding * $rate;
      $totalProvision += $provision;

      $results[] = [
        'loan_id' => $loan->id,
        'loan_number' => $loan->loan_number,
        'member' => $loan->member->full_name,
        'outstanding' => $outstanding,
        'days_past_due' => $daysPastDue,
        'classification' => match(true) { $daysPastDue <=30 => 'Performing', $daysPastDue <=60 => 'Pass & Watch', $daysPastDue <=90 => 'Substandard', $daysPastDue <=180 => 'Doubtful', default => 'Lost' },
        'rate' => $rate,
        'provision' => $provision
      ];
    }

    // Post total provision to ledger
    // Get existing provision balance, calculate difference, post adjustment
    // LedgerService::post('5004','1205', $difference, $provisionBatch, "Loan Loss Provision ".now()->format('M Y"))

    return ['total' => $totalProvision, 'details' => $results];
  }
}
